î°ƒimport { Request, Response, NextFunction } from 'express';
import { supabase } from '../lib/supabase';
import { Redis } from 'ioredis';
import jwt from 'jsonwebtoken';

const redis = new Redis(process.env.REDIS_URL || 'redis://localhost:6379');

export interface AuthenticatedTenantRequest extends Request {
  tenantId?: string;
  dealId?: string;
  dealState?: any;
  isCommercialOpportunity?: boolean;
}

export async function gatekeeperMiddleware(
  req: AuthenticatedTenantRequest,
  res: Response,
  next: NextFunction
) {
  try {
    const authHeader = req.headers.authorization;
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({ error: 'Unauthorized: Missing or invalid JWT authorization header' });
    }
    const token = authHeader.split(' ')[1];
    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'fallback-secret') as { tenant_id: string };
    const { deal_id } = req.body;
    const tenant_id = decoded.tenant_id;

    if (!deal_id || !tenant_id) {
      return res.status(400).json({ error: 'Missing required deal_id or tenant_id claims' });
    }

    // Distributed Redis Lock for Idempotency & Race Protection
    const lockKey = `lock:deal:${tenant_id}:${deal_id}`;
    const acquired = await redis.set(lockKey, 'locked', 'NX', 'EX', 30);
    if (!acquired) {
      return res.status(429).json({ error: 'Concurrent engine execution detected for this deal. Request locked.' });
    }

    // RLS Scoped Database Query
    const { data: deal, error } = await supabase
      .from('deals')
      .select('id, status, stage, last_processed_at, tenant_id, deal_type, commercial_intent_score')
      .eq('id', deal_id)
      .eq('tenant_id', tenant_id)
      .single();

    if (error || !deal) {
      await redis.del(lockKey);
      return res.status(404).json({ error: 'Deal not found or tenant RLS restriction failed' });
    }

    // Block non-executable status
    const BLOCKED_STATUSES = ['CLOSED_WON', 'CLOSED_LOST', 'PAUSED', 'OPTED_OUT', 'ARCHIVED'];
    if (BLOCKED_STATUSES.includes(deal.status.toUpperCase())) {
      await redis.del(lockKey);
      return res.status(422).json({ error: `Execution rejected: Deal status is '${deal.status}'` });
    }

    // Gate 0: Commercial Reality & Intent Filter (Omega Filter)
    if (deal.deal_type === 'PEER_NETWORKING' || (deal.commercial_intent_score && deal.commercial_intent_score < 20)) {
      await redis.del(lockKey);
      return res.status(200).json({
        status: 'NOT_A_DEAL',
        riskIndex: 0,
        actionTrigger: 'SUPPRESS_OUTREACH',
        reason: 'Gate 0 Reality Filter: Non-commercial interaction or low opportunity score.'
      });
    }

    req.tenantId = tenant_id;
    req.dealId = deal_id;
    req.dealState = deal;
    req.isCommercialOpportunity = true;

    next();
  } catch (err: any) {
    return res.status(500).json({ error: 'Gatekeeper middleware internal error', details: err.message });
  }
}

export interface EngineMatrixDecision {
  targetStage: 'STALLED' | 'RE_ENGAGED' | 'ESCALATED' | 'REQUIRES_MANUAL_REVIEW';
  riskIndex: number;
  actionTrigger: 'SEND_REENGAGEMENT_EMAIL' | 'SEND_EXECUTIVE_BUMP' | 'PAUSE_WORKFLOW' | 'SUPPRESS_OUTREACH';
  templateId: string;
  reprofiledStakeholders: StakeholderProfile[];

    deltaVelocity: number;
    phiEntropy: number;
    psiSentiment: number;
    gammaStakeholder: number;
  };
}

export class DeterministicEngineMatrix {
  public static evaluate(payload: DealInputPayload): EngineMatrixDecision {
    const {
      daysStalled,
      stageBenchmarkDays = 14,
      unansweredEmailCount,
      lastProspectSentiment,
      contractValue,
      stakeholders
    } = payload;

    // 1. Delta_Velocity = 100 * (1 - exp(-0.85 * (daysStalled / stageBenchmarkDays)))
    const velocityRatio = daysStalled / Math.max(1, stageBenchmarkDays);
    const deltaVelocity = 100 * (1 - Math.exp(-0.85 * velocityRatio));

    // 2. Phi_Entropy = 35 * (1 - exp(-0.5 * unansweredEmailCount))
    const phiEntropy = 35 * (1 - Math.exp(-0.5 * unansweredEmailCount));

    // 3. Psi_Sentiment weights: POSITIVE: 0, NEUTRAL: 12, GHOSTING: 25, NEGATIVE: 40
    const sentimentWeights: Record<string, number> = {
      POSITIVE: 0,
      NEUTRAL: 12,
      GHOSTING: 25,
      NEGATIVE: 40
    };
    const psiSentiment = sentimentWeights[lastProspectSentiment] ?? 25;

    // Dynamic Stakeholder Intelligence Re-Profiling
    const reprofiledStakeholders = stakeholders.map(s => {
      if (s.role === 'CHAMPION' && unansweredEmailCount >= 3) {
        return { ...s, role: 'DETRACTOR' as const, lastObservedSignal: 'Champion un-engaged across 3+ attempts' };
      }
      return s;
    });

    // 4. Gamma_Stakeholder = 15 * (riskStakeholders / totalStakeholders)
    const totalStakeholders = Math.max(1, reprofiledStakeholders.length);
    const riskStakeholders = reprofiledStakeholders.filter(s => s.role === 'BLOCKER' || s.role === 'DETRACTOR').length;
    const gammaStakeholder = 15 * (riskStakeholders / totalStakeholders);

    // DRI Formula Calculation
    const rawDRI = deltaVelocity + phiEntropy + psiSentiment + gammaStakeholder;
    const riskIndex = Math.min(100, Math.max(0, Math.round(rawDRI)));

    const breakdown = {
      deltaVelocity: Math.round(deltaVelocity * 100) / 100,
      phiEntropy: Math.round(phiEntropy * 100) / 100,
      psiSentiment,
      gammaStakeholder: Math.round(gammaStakeholder * 100) / 100
    };

    if (riskIndex >= 85 || unansweredEmailCount >= 4) {
      return {
        targetStage: 'REQUIRES_MANUAL_REVIEW',
        riskIndex,
        actionTrigger: 'PAUSE_WORKFLOW',
        templateId: 'MANUAL_ESCALATION_V1',
        reprofiledStakeholders,
        breakdown
      };
    }
    if (daysStalled >= 14 && contractValue >= 25000) {
      return {
        targetStage: 'ESCALATED',
        riskIndex,
        actionTrigger: 'SEND_EXECUTIVE_BUMP',
        templateId: 'EXEC_BREAKTHROUGH_V2',
        reprofiledStakeholders,
        breakdown
      };
    }
    return {
      targetStage: 'RE_ENGAGED',
      riskIndex,
      actionTrigger: 'SEND_REENGAGEMENT_EMAIL',
      templateId: 'SOFT_REENGAGE_V1',
      reprofiledStakeholders,
      breakdown
    };
  }
}

      });
      const completion = await ollama.chat.completions.create({
        model: process.env.OLLAMA_MODEL || 'qwen2.5:7b',
        messages: [{ role: 'user', content: prompt }],
        response_format: { type: 'json_object' }
      });
      const content = completion.choices[0]?.message?.content || '{}';
      return AIResponseSchema.parse(JSON.parse(content));
    } else {
      // Production Branch: Gemini 1.5 Pro
      const apiKey = process.env.GEMINI_API_KEY;
      if (!apiKey) return getContextAwareFallback(templateId, variables);

      const ai = new GoogleGenerativeAI(apiKey);
      const model = ai.getGenerativeModel({
        model: 'gemini-1.5-pro',
        generationConfig: {
          responseMimeType: 'application/json',
          responseSchema: {
            type: SchemaType.OBJECT,
            properties: {
              subjectLine: { type: SchemaType.STRING },
              emailBody: { type: SchemaType.STRING },
              callToAction: { type: SchemaType.STRING },
              detectedObjection: { type: SchemaType.STRING },
              evidentiaryExcerpts: { type: SchemaType.ARRAY, items: { type: SchemaType.STRING } }
            },
            required: ['subjectLine', 'emailBody', 'callToAction', 'detectedObjection', 'evidentiaryExcerpts']
          }
        }
      });
      const response = await model.generateContent(prompt);
      return AIResponseSchema.parse(JSON.parse(response.response.text()));
    }
  } catch (error) {
    console.error('Circuit-breaker activated in AI adaptability layer:', error);
    return getContextAwareFallback(templateId, variables);
  }
}

export class AdvancedDeterministicEngineMatrix {
  public static evaluate(payload: RefinedDealInputPayload) {
    const {
      tStalled,
      muStage = 14,
      sigmaStage = 5,
      unansweredCount,
      lastProspectSentiment,
      daysSinceSentimentSignal = 0,
      stakeholders
    } = payload;

    // 1. Stage Duration Volatility Z-Score & Delta Velocity Decay
    const zStall = Math.max(0, (tStalled - muStage) / Math.max(1, sigmaStage));
    const lambda = 0.85;
    const deltaVelocity = 100 * (1 - Math.exp(-lambda * zStall));

    // 2. Touchpoint Entropy Penalty
    const beta = 35;
    const alpha = 0.5;
    const phiEntropy = beta * (1 - Math.exp(-alpha * unansweredCount));

    // 3. Sentiment Memory Decay ( = 14 days half-life)
    const baseSentimentWeights: Record<string, number> = {
      POSITIVE: 0,
      NEUTRAL: 12,
      GHOSTING: 25,
      NEGATIVE: 40
    };
    const psiBase = baseSentimentWeights[lastProspectSentiment] ?? 25;
    const tau = 14;
    const psiSentiment = psiBase * Math.exp(-(Math.LN2 / tau) * daysSinceSentimentSignal);

    // 4. Stakeholder Authority Power Law (W_i)
    const authorityWeights: Record<string, number> = {
      ECONOMIC_BUYER: 0.45,
      CHAMPION: 0.30,
      BLOCKER: 0.15,
      DETRACTOR: 0.15,
      USER: 0.10
    };
    const riskStakeholders = stakeholders.filter(s => s.role === 'BLOCKER' || s.role === 'DETRACTOR');
    const sumWeights = riskStakeholders.reduce((acc, s) => acc + (authorityWeights[s.role] || 0.10), 0);
    const gammaStakeholder = 15 * sumWeights;

    // Final Refined DRI Formula
    const rawDRI = deltaVelocity + phiEntropy + psiSentiment + gammaStakeholder;
    const riskIndex = Math.min(100, Math.max(0, Math.round(rawDRI)));

    return {
      riskIndex,
      breakdown: {
        deltaVelocity: Math.round(deltaVelocity * 100) / 100,
        phiEntropy: Math.round(phiEntropy * 100) / 100,
        psiSentiment: Math.round(psiSentiment * 100) / 100,
        gammaStakeholder: Math.round(gammaStakeholder * 100) / 100
      }
    };
  }
}
